import sqlite3
import pandas as pd
import numpy as np
import os

os.makedirs("data/clean", exist_ok=True)
os.makedirs("outputs", exist_ok=True)

Load raw exports with keep_default_na=False to preserve empty strings
customers = pd.read_csv("data/raw/customers.csv", keep_default_na=False)
products = pd.read_csv("data/raw/products.csv", keep_default_na=False)
orders = pd.read_csv("data/raw/orders.csv", keep_default_na=False)

--- Clean Customers ---
customers['city'] = customers['city'].str.strip().str.title()
customers['gender'] = customers['gender'].apply(lambda g: g if g in ['male', 'female'] else 'unknown')

Fill missing ages with median age of customer's city
median_ages = customers[customers['age'] > 0].groupby('city')['age'].median()
customers['age'] = customers.apply(lambda row: median_ages[row['city']] if row['age'] == "" or row['age'] < 10 or row['age'] > 100 else row['age'], axis=1)
customers['age'] = customers['age'].astype(int)
customers['signup_date'] = pd.to_datetime(customers['signup_date'])

--- Clean Orders ---
orders['order_ts'] = pd.to_datetime(orders['order_ts'])
orders = orders.drop_duplicates()
Keep latest record for duplicate order_ids (pending vs final)
orders = orders.sort_values('order_ts').drop_duplicates(subset=['order_id'], keep='last')

Convert order_amount and handle missing values using product catalogue prices
prod_map = products.set_index('product_id')[['price_inr']].to_dict()['price_inr']
def fix_amount(row):
    try:
        val = float(row['order_amount'])
        if val < 0: raise ValueError
        return val
    except:
        price = prod_map.get(row['product_id'], 100)
        return round(price * row['quantity'] * (1 - row['discount_pct'] / 100), 2)

orders['order_amount'] = orders.apply(fix_amount, axis=1)
orders.loc[orders['delivery_minutes'] < 0, 'delivery_minutes'] = np.nan

Impute delivery minutes using median of product category for completed orders
completed = orders[orders['status'] == 'completed'].merge(products[['product_id', 'category']], on='product_id')
cat_medians = completed.groupby('category')['delivery_minutes'].median()
orders = orders.merge(products[['product_id', 'category']], on='product_id', how='left')
orders['delivery_minutes'] = orders.apply(
    lambda r: cat_medians[r['category']] if pd.isna(r['delivery_minutes']) and r['status'] == 'completed' else r['delivery_minutes'], axis=1
)
orders.drop(columns=['category'], inplace=True)

Save cleaned CSVs
customers.to_csv("data/clean/customers.csv", index=False)
products.to_csv("data/clean/products.csv", index=False)
orders.to_csv("data/clean/orders.csv", index=False)

Store into SQLite Database
conn = sqlite3.connect("data/instamart.db")
customers.to_sql("customers", conn, if_exists="replace", index=False)
products.to_sql("products", conn, if_exists="replace", index=False)
orders.to_sql("orders", conn, if_exists="replace", index=False)
conn.close()

3. SQL Analysis Queries (sql/queries.sql)
-- Q1: WHERE then HAVING
SELECT customer_id, SUM(order_amount) as total_spend
FROM orders WHERE status = 'completed' AND order_amount > 500
GROUP BY customer_id HAVING total_spend > 8000;

-- Q2: Three-table join
SELECT c.city, p.category, SUM(o.order_amount) as revenue
FROM orders o JOIN customers c ON o.customer_id = c.customer_id JOIN products p ON o.product_id = p.product_id
WHERE o.status = 'completed' GROUP BY c.city, p.category;

-- Q3: Scalar subquery
SELECT * FROM products WHERE price_inr > (SELECT AVG(price_inr) FROM products);

-- Q4: LEFT JOIN
SELECT c.customer_id FROM customers c LEFT JOIN orders o ON c.customer_id = o.customer_id AND o.status = 'completed' WHERE o.order_id IS NULL;

-- Q5: CTE
WITH cust_spend AS (SELECT customer_id, SUM(order_amount) as total FROM orders WHERE status = 'completed' GROUP BY customer_id)
SELECT * FROM cust_spend WHERE total > (SELECT AVG(total) FROM cust_spend);

-- Q6: Ranking window functions
SELECT customer_id, city, total_spend, RANK() OVER(PARTITION BY city ORDER BY total_spend DESC) as rnk FROM ...;

-- Q7: Running total
SELECT SUBSTR(order_ts, 1, 10) as order_date, SUM(order_amount) OVER(ORDER BY order_ts) as running_rev FROM orders WHERE status = 'completed';

from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import json
import glob

Chunking policies
chunks = []
for filepath in glob.glob("data/policies/*.txt"):
    with open(filepath, 'r', encoding='utf-8') as f:
        text = f.read()
    words = text.split()
    for i in range(0, len(words), 30): 40 words with 10 overlap
        chunk = " ".join(words[i:i+40])
        chunks.append({'source': filepath, 'content': chunk})

vectorizer = TfidfVectorizer().fit([c['content'] for c in chunks])
chunk_vectors = vectorizer.transform([c['content'] for c in chunks])

def retrieve(query, k=3):
    qv = vectorizer.transform([query])
    scores = cosine_similarity(qv, chunk_vectors).flatten()
    top_indices = scores.argsort()[-k:][::-1]
    return [chunks[idx] for idx in top_indices]
